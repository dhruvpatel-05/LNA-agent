\section{Motivation}
\begin{frame}{Hypothesis}
\begin{block}{Core idea}
OFI reveals \emph{latent demand}; LLM sentiment reveals the
\emph{direction} of a news shock.
Their interaction should predict short-horizon returns better than either alone.
\end{block}
\vspace{1em}
\[ \text{signal}_t = \text{OFI}_t \times \text{sent}_t \]
Evaluated at 1, 5, 15-minute forward return horizons via Spearman IC.
\end{frame}

\begin{frame}{Conditional OFI Regression}
\[
r_{t+h} = \alpha + \gamma\,\text{OFI} + \delta_1\,s^+ + \delta_2\,s^-
+ \beta^+(\text{OFI}\cdot s^+) + \beta^-(\text{OFI}\cdot s^-)
+ \varepsilon
\]
\begin{itemize}
  \item $s^+ = \max(s,0)$, $s^- = \min(s,0)$ (hinge split)
  \item Newey--West HAC SEs for overlapping labels
  \item Wald test: $H_0\colon\beta^+=\beta^-$ (tone symmetry)
\end{itemize}
\end{frame}
